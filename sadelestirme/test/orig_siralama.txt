ee=Getiri("s2h","TL")>Getiri("s2h","TL","XUTUM")-10;
f=NetSatisBuyumeYillik()>Tufe(12);
gx=FAVOKBuyumeYillik()>Tufe(12) or (FAVOK("TL",0)>FAVOK("TL",-1) and FAVOK("TL",0)>FAVOK("TL",-4));
hx=(BrutEFKBuyumeYillik()>Tufe(12) or (BrutEFK("TL",0)>BrutEFK("TL",-1) and BrutEFK("TL",0)>BrutEFK("TL",-4))) and (EFKBuyumeYillik()>Tufe(12) or (EFK("TL",0)>EFK("TL",-1) and EFK("TL",0)>EFK("TL",-4)));
efkTeyit=EFK("TL",0)>=EFK("TL",-1) or EFK("TL",0)>=EFK("TL",-4);
tam=IF(ee and f and (hx or gx) and efkTeyit,1,0);
xE=((EFK("TL",0)-EFK("TL",-4))/Abs(EFK("TL",-4)))/0.7;
xF=((FAVOK("TL",0)-FAVOK("TL",-4))/Abs(FAVOK("TL",-4)))/0.6;
xR=(OzsermayeKarlilikYillik()-OzsermayeKarlilikYillik("",-1))/2;
xI=(EFKBuyumeYillik()-EFKBuyumeYillik("",-1))/10;
zRaw=Teknik.Indicator("C/HHV(H,252)","d");
xZ=(zRaw-1)/0.0865;
pE=IF(xE == null, -1, xE/(1+Abs(xE)));
pF=IF(xF == null, -1, xF/(1+Abs(xF)));
pR=IF(xR == null, -1, xR/(1+Abs(xR)));
pI=IF(xI == null, -1, xI/(1+Abs(xI)));
pZ=IF(zRaw == null, 0, xZ/(1+Abs(xZ)));
r75Raw=Teknik.Indicator("C/Mov(C,75,S)","d");
pen75=IF(r75Raw == null, 0, IF(r75Raw>1.35, -15, 0));
m6Raw=Teknik.Indicator("Ref(C,-21)/Ref(C,-126)","d");
m6x=m6Raw-1;
m6c=IF(m6x>1, 1, IF(m6x<-1, -1, m6x));
mom6=IF(m6Raw == null, 0, 8*m6c);
karIvmesi=100+25*(pE+pF+pR+pI+pZ)+pen75+mom6;
SKOR=tam*1000+karIvmesi;
